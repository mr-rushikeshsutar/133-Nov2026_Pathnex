 name: nginx-deployment

  matchLabels :
   app: pathnex

   labels:
    app: pathnex

  spec:
   containers:
    - name: web
      image: nginx
      ports:
       - containerPort: 80

#shell-script - Print CPU, Memory, Disk

#!/bin/bash
CPU_USAGE=$(top -bn1 | grep "Cpu(s)" | awk '{print 100 - $8}')
 
echo "CPU Usage: ${CPU_USAGE}%"
echo "memory usage: "
free -h
echo "disk usgae"
df -h /
