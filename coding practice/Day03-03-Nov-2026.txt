#nodejs build-jenkins

pipeline {
    agent any
    tools{
        nodejs 'NodeJS-16'
    }

    stages{
        stage('checkout'){
            steps{
                git branch 'main', url: 'https://github.com/Pathnex/sample-node-app.git'
            }
        }
        stage('install-dependencies'){
            steps{
                sh 'npm install'
            }
        }
        stage('run test'){
            steps{
                sh 'npm test'
            }
        }
        stage('build'){
            steps{
                sh 'npm run build'
            }
        }
    }
}
