#jenkins - Archive Artifacts
pipeline {
agent any 
environment {
INSTITUTE_NAME = "pathnex"
}
stages{
stage('checkout') {
steps{
git branch 'main', url:'https://github.com/Pathnex/sample-java-app.git'
}
}

stage('build') {
steps{
sh 'mvn clean package'
}
}

stage('archive'){
steps{
 archiveArtifacts artifacts: 'target/.*jar', fingerptint: true
}
}

}

}


#gitlab - Archive Artifacts

stages:
 - build
 - deploy

 - cd sample-java-app
 - mvn clean package

 artifacts:
  paths:
  - target/.*jar
  expire_in: 1 week
