 name: pathnex-rolling

 replicas: 3

  type: RollingUpdate
  rollingUpdate:
   maxSurge: 1
   maxUnavailable: 1
 selector:
  matchLabels:
   app: roll
 template:
  metadata:
   labels: 
    app: roll
  spec:
   containers:
   - name: web
     image: nginx


#shell-script - Backup Directory

#!/bin/bash
source="/var/log"
dest ="/backup/pathnex-$(date + %F).tar.gz"


mkdir -p /backup

tar -czf $dest $source 
echo "backup created at : $dest"

#jenkins - Multi-branch Pipelines

pipeline {
agent any
environment{
INSTITUTE_NAME = "pathnex"
}
stages{
stage('checkout'){
steps{
checkout scm 
}
}

stage('build'){
steps{
sh 'mvn clean package'
}
}
}
}



#gitlab - Multi-branch Pipelines

stages:
- build

build-main:
 stage: build 
 image: mvn:3.8.9-eclips-temurin-17
 script:
 - git clone 
 - cd 
 - mvn clean package
 rules:
 - if : "$CI_COMMIT_BRANCH === "main""

build-dev:
 stage: build
 image: mvn:3.8.9-eclips-temurin-17
 script:
 - mvn clean package
 rules :
 - if : "$CI_COMMIT_BRANCH ===="dev""
