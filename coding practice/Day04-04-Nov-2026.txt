#jenkins-parallel build-test

pipeline {
    agent any
    stages{
        stage('checkout'){
            steps {
                git branch 'main', https://github.com/Pathnex/sample-java-app.git
            }
        }
        stage ('build-test-parallel'){

            parallel {
                stage('build'){
                   steps{
                    sh 'mvn clean compile'
                   }
                }
                stage('test'){
                    steps{
                       sh 'mvn test'
                    }
                }
            }    
        }
        
    }
}
