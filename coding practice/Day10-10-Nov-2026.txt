#jenkins - Cache Maven Dependencies

pipeline {
agent any 
environment {
INSTITUTE_NAME = "pathnex"
}
stages{

stage ('checkout'){
 steps{
 git branch 'main', url:'https://github.com/Pathnex/sample-java-app.git' 
 }
}

stage('build') {
steps {
sh '''
  if [-d ".m2"]; then
    echo "using cached dependencies"

else 
 mvn clean compile
fi 
...

}
}
}

}
